-- TABLA 5: VENTAS_DETALLE (Soporta E3HU1, E3HU2)

PRODUCTO_ID NUMBER NOT NULL,

CONSTRAINT CHK_CANTIDAD_DET CHECK (CANTIDAD > 0),
CONSTRAINT CHK_SUBTOTAL CHECK (SUBTOTAL > 0)

-- 5. TRIGGER PL/SQL: ACTUALIZACIÓN Y AUDITORÍA DE STOCK EN TIEMPO REAL

CREATE OR REPLACE TRIGGER TRG_DESCONTAR_STOCK
BEFORE INSERT ON VENTAS_DETALLE
FOR EACH ROW
DECLARE
v_stock_actual NUMBER;
BEGIN
-- 1. Validar el stock disponible antes de la inserción del detalle
SELECT CANTIDAD_STOCK INTO v_stock_actual
FROM PRODUCTOS
WHERE PRODUCTO_ID = :NEW.PRODUCTO_ID
FOR UPDATE;

IF v_stock_actual < :NEW.CANTIDAD THEN
    RAISE_APPLICATION_ERROR(-20001, 'Error de Stock: No hay suficiente stock disponible para el producto ID ' || :NEW.PRODUCTO_ID);
END IF;

-- 2. Descontar las unidades vendidas
UPDATE PRODUCTOS
SET CANTIDAD_STOCK = CANTIDAD_STOCK - :NEW.CANTIDAD,
    ESTADO = CASE 
                WHEN (CANTIDAD_STOCK - :NEW.CANTIDAD) = 0 THEN 'AGOTADO' 
                ELSE 'DISPONIBLE' 
             END
WHERE PRODUCTO_ID = :NEW.PRODUCTO_ID;


END;
/

-- 6. PROCEDIMIENTO ALMACENADO PL/SQL: REGISTRO ATÓMICO DE VENTA (E3HU1)

CREATE OR REPLACE PROCEDURE SP_REGISTRAR_VENTA (
p_empleado_id    IN NUMBER,
p_medio_pago     IN VARCHAR2,
p_producto_id    IN NUMBER,
p_cantidad       IN NUMBER,
p_venta_id       OUT NUMBER,
p_mensaje_res    OUT VARCHAR2
) AS
v_precio_unitario NUMBER(10,2);
v_subtotal        NUMBER(10,2);
BEGIN
-- Validar precio del producto
SELECT PRECIO_VENTA INTO v_precio_unitario
FROM PRODUCTOS
WHERE PRODUCTO_ID = p_producto_id;

v_subtotal := v_precio_unitario * p_cantidad;

-- 1. Insertar la cabecera de la venta
INSERT INTO VENTAS_CABECERA (EMPLEADO_ID, MEDIO_PAGO, TOTAL_VENTA)
VALUES (p_empleado_id, UPPER(p_medio_pago), v_subtotal)
RETURNING VENTA_ID INTO p_venta_id;

-- 2. Insertar el detalle (Esto dispara automáticamente el TRG_DESCONTAR_STOCK)
INSERT INTO VENTAS_DETALLE (VENTA_ID, PRODUCTO_ID, CANTIDAD, PRECIO_UNITARIO, SUBTOTAL)
VALUES (p_venta_id, p_producto_id, p_cantidad, v_precio_unitario, v_subtotal);

-- Confirmar transacción
COMMIT;
p_mensaje_res := 'Venta registrada exitosamente. Comprobante ID: ' || p_venta_id;


EXCEPTION
WHEN OTHERS THEN
ROLLBACK;
p_mensaje_res := 'Error al procesar la venta: ' || SQLERRM;
END SP_REGISTRAR_VENTA;
/

-- Consulta de Cuadre/Cierre Diario de Caja (E3HU2)
SELECT
vc.VENTA_ID AS "ID Venta",
TO_CHAR(vc.FECHA_VENTA, 'DD/MM/YYYY HH24:MI:SS') AS "Fecha y Hora",
emp.NOMBRE AS "Cajero Responsable",
vc.MEDIO_PAGO AS "Medio Pago",
TO_CHAR(vc.TOTAL_VENTA, '$999,999,999.00') AS "Monto Total"
FROM VENTAS_CABECERA vc
JOIN USUARIOS_EMPLEADOS emp ON vc.EMPLEADO_ID = emp.EMPLEADO_ID
WHERE TRUNC(vc.FECHA_VENTA) = TRUNC(SYSDATE)
ORDER BY vc.VENTA_ID DESC;
